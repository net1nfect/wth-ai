def _parse_timeout(raw: object, *, default: int = 30, maximum: int = 300) -> int:
    """Timeout request RPC: angka positif, dipotong ke `maximum`."""
    if raw is None:
        return default
    try:
        value = int(raw)
    except (TypeError, ValueError):
        raise ValueError(f"timeout tidak valid: {raw!r}") from None
    if value <= 0:
        raise ValueError(f"timeout harus > 0: {value}")
    return min(value, maximum)
```

def _vtuple(text: str) -> tuple[int, ...]:
    parts = re.findall(r"\d+", str(text))
    return tuple(int(p) for p in parts) or (0,)


def _version_in_range(cve: dict[str, Any], version: str) -> bool:
    """Applicability kasar: bandingkan komponen numerik versi."""
    matches = [m
               for cfg in cve.get("configurations") or []
               for node in cfg.get("nodes") or []
               for m in node.get("cpeMatch") or []]
    if not matches:
        return True
    have = _vtuple(version)
    for m in matches:
        lo_in = m.get("versionStartIncluding")
        lo_ex = m.get("versionStartExcluding")
        hi_in = m.get("versionEndIncluding")
        hi_ex = m.get("versionEndExcluding")
        if lo_in and have < _vtuple(lo_in):
            continue
        if lo_ex and have <= _vtuple(lo_ex):
            continue
        if hi_in and have > _vtuple(hi_in):
            continue
        if hi_ex and have >= _vtuple(hi_ex):
            continue
        return True
    return False
```

import fcntl
from contextlib import contextmanager


    @contextmanager
    def locked(self):
        lock_path = self.path.with_name(self.path.name + ".lock")

def _valid_http_args(args: dict[str, Any]) -> str:
    url = args.get("url")
    if not isinstance(url, str) or not url.startswith(("http://", "https://")):
        return f"url HTTP tidak valid: {url!r}"
    headers = args.get("headers")
    if headers is not None and not (
